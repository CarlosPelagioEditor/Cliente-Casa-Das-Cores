      if (db) {

          .then((doc) => {

    /* =========================================================================
       6. CALCULADORA DE TINTAS COM FÓRMULA DINÂMICA
       ========================================================================= */
    function calcularTinta() {
      const larg = parseFloat(document.getElementById("calc-largura").value) || 0;
      const alt = parseFloat(document.getElementById("calc-altura").value) || 0;
      const dem = parseInt(document.getElementById("calc-demaos").value) || configCalculadora.demaosPadrao || 2;
      const port = parseFloat(document.getElementById("calc-portas").value) || 0;

      const areaTotal = Math.max(0, (larg * alt) - port);

      // FÓRMULA DINÂMICA: Litros Necessários = (Área Total / Rendimento por Litro) × Demãos
      const rendimento = configCalculadora.rendimentoLitro || 10;
      const litros = ((areaTotal / rendimento) * dem).toFixed(1);

      document.getElementById("res-area").innerText = areaTotal.toFixed(1);
      document.getElementById("res-litros").innerText = litros;

      const rec = document.getElementById("res-latas");
      const numLitros = parseFloat(litros);

      if (numLitros <= 0) {
        rec.innerText = "Informe medidas válidas da parede.";
      } else if (numLitros <= 1) {
        rec.innerText = "Recomendação: 1 Quarto de 900ml cobre esta área!";
      } else if (numLitros <= 4) {
        rec.innerText = "Recomendação: 1 Galão de 3,6 Litros cobre com folga!";
      } else if (numLitros <= 7.5) {
        rec.innerText = "Recomendação: 2 Galões de 3,6 Litros (total 7,2L)!";
      } else {
        const latas18 = Math.ceil(numLitros / 18);
        rec.innerText = `Recomendação: ${latas18} Lata(s) de 18 Litros para máxima economia!`;
      }
    }

    function loginGoogle() {
      if (!auth) {
        showToast("Carregando serviço de autenticação...");
        return;
      }

      if (usuarioGoogle) {
        auth.signOut().then(() => {
          showToast("Você saiu da sua conta Google.");
        });
      } else {
        const provider = new firebase.auth.GoogleAuthProvider();
        auth.signInWithPopup(provider)
          .then(() => {
            showToast("Login Google efetuado com sucesso!");
          })
          .catch((err) => {
            console.warn("Aviso Login Google:", err?.code || err?.message);
            showToast("Use o botão 'Entrar como Administrador' com seu usuário e senha.");
          });
      }
    }
